cmake_minimum_required(VERSION 3.16)
project(wifi_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
find_package(Qt5 REQUIRED COMPONENTS Widgets DBus Test)
enable_testing()
add_executable(wifi_tests wifi_test.cpp ../wifimanager.cpp ../formwifi.cpp)
target_include_directories(wifi_tests PRIVATE ..)
target_link_libraries(wifi_tests PRIVATE Qt5::Widgets Qt5::DBus Qt5::Test)
add_test(NAME wifi COMMAND dbus-run-session -- sh -c "export DBUS_SYSTEM_BUS_ADDRESS=\"\$DBUS_SESSION_BUS_ADDRESS\"; export QT_QPA_PLATFORM=offscreen; exec \"$<TARGET_FILE:wifi_tests>\"")
