cmake_minimum_required(VERSION 3.16)

set(CMAKE_AUTOMOC ON)

find_package(Qt5 REQUIRED COMPONENTS Widgets Test)
enable_testing()

target_link_libraries(time_input_tests PRIVATE Qt5::Widgets Qt5::Test)
add_test(NAME time_input COMMAND time_input_tests)
set_tests_properties(time_input PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
