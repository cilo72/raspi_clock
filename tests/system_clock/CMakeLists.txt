cmake_minimum_required(VERSION 3.16)
project(system_clock_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt5 REQUIRED COMPONENTS Core DBus Test)
enable_testing()
add_executable(system_clock_tests ../system_clock_test.cpp ../../systemclock.cpp)
target_include_directories(system_clock_tests PRIVATE ../..)
target_link_libraries(system_clock_tests PRIVATE Qt5::Core Qt5::DBus Qt5::Test)
add_test(NAME system_clock COMMAND dbus-run-session -- "$<TARGET_FILE:system_clock_tests>")
set_tests_properties(system_clock PROPERTIES TIMEOUT 30 ENVIRONMENT "TZ=Europe/Zurich")
